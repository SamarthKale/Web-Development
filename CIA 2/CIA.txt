Q1 — Dynamic To-Do List with jQuery                              [BROWSER]
────────────────────────────────────────────────────────────────────────────────

PROBLEM:
  Create a To-Do List where users can:
  - Add new tasks
  - Delete tasks via a Remove button
  - Toggle task completion (strike-through) on click
  Use jQuery event delegation for dynamically added tasks.
  Apply fadeIn() when adding and fadeOut() when removing tasks.

FILE: todo.html

EXECUTION:
  Save the file and open in any browser (Chrome, Firefox, Edge).
  No server required — works directly from the file system.

CODE:
──────
<!DOCTYPE html>
<html lang="en">
<head>
<meta charset="UTF-8">
<title>Dynamic To-Do List</title>
<script src="https://code.jquery.com/jquery-3.6.0.min.js"></script>
<style>
  body { font-family: Arial, sans-serif; max-width: 500px; margin: 40px auto; padding: 0 15px; }
  h2 { color: #333; }
  #taskInput { padding: 10px; width: 65%; border: 1px solid #ccc; border-radius: 4px; }
  #addBtn { padding: 10px 18px; background: #4CAF50; color: white; border: none; border-radius: 4px; cursor: pointer; }
  #addBtn:hover { background: #45a049; }
  .task { padding: 12px; margin: 8px 0; background: #f4f4f4; border-left: 4px solid #4CAF50;
          cursor: pointer; display: flex; justify-content: space-between; align-items: center; }
  .completed { text-decoration: line-through; opacity: 0.6; border-left-color: #999; }
  .remove { color: #e74c3c; font-weight: bold; cursor: pointer; padding: 0 8px; font-size: 18px; }
  .remove:hover { color: #c0392b; }
</style>
</head>
<body>
  <h2>My To-Do List</h2>
  <input type="text" id="taskInput" placeholder="Enter a new task...">
  <button id="addBtn">Add Task</button>
  <div id="taskList"></div>
<script>
$(document).ready(function () {
  $('#addBtn').click(function () {
    var taskText = $('#taskInput').val().trim();
    if (taskText === '') { alert('Please enter a task.'); return; }
    var $task = $('<div class="task"><span class="text"></span><span class="remove">&times;</span></div>');
    $task.find('.text').text(taskText);
    $task.hide().appendTo('#taskList').fadeIn(400);
    $('#taskInput').val('').focus();
  });
  $('#taskInput').keypress(function (e) { if (e.which === 13) $('#addBtn').click(); });
  $('#taskList').on('click', '.task', function (e) {
    if (!$(e.target).hasClass('remove')) $(this).toggleClass('completed');
  });
  $('#taskList').on('click', '.remove', function (e) {
    e.stopPropagation();
    $(this).closest('.task').fadeOut(400, function () { $(this).remove(); });
  });
});
</script>
</body>
</html>

  var $slides = $('.slide'), current = 0, total = $slides.length, autoPlay;
  function showSlide(index) {
    $slides.fadeOut(500); $slides.eq(index).fadeIn(500);
    $('#indicator').text('Slide ' + (index + 1) + ' of ' + total);

  function prevSlide() { current = (current - 1 + total) % total; showSlide(current); }

  function stopAutoPlay() { clearInterval(autoPlay); }

  $('#slider').hover(stopAutoPlay, startAutoPlay);
  showSlide(current); startAutoPlay();
});
</script>

Q3 — Real-Time Username Availability Checker                     [BROWSER]

  Develop a signup form where as the user types a username:
  - An AJAX request checks if it's available
  - Display "Available" (green) or "Taken" (red) without page reload
  - Use jQuery $.post() (simulated) and slideDown() to show the status message

  Save the file and open in any browser.
  No server required. Try usernames like "admin", "john", or "test" to see Taken response.

CODE:
──────
<!DOCTYPE html>
<html lang="en">
<head>
<meta charset="UTF-8">
<title>Username Availability Checker</title>
<script src="https://code.jquery.com/jquery-3.6.0.min.js"></script>
<style>
  body { font-family: Arial; max-width: 420px; margin: 60px auto; padding: 20px;
         background: #f9f9f9; border: 1px solid #ddd; border-radius: 6px; }
  h2 { color: #333; }
  input { width: 100%; padding: 10px; box-sizing: border-box; font-size: 14px;
          border: 1px solid #ccc; border-radius: 4px; }
  #status { display: none; margin-top: 10px; padding: 10px; border-radius: 4px; font-weight: bold; }
  .available { background: #d4edda; color: #155724; }
  .taken { background: #f8d7da; color: #721c24; }
  .checking { background: #fff3cd; color: #856404; }
</style>
</head>
<body>
  <h2>Sign Up</h2>
  <label>Username:</label>
  <input type="text" id="username" placeholder="Type a username...">
  <div id="status"></div>
<script>
$(document).ready(function () {
  var takenUsernames = ['admin', 'john', 'test', 'user', 'sann', 'mary'];
  function checkUsername(name, callback) {
    setTimeout(function () {
      callback({ available: takenUsernames.indexOf(name.toLowerCase()) === -1 });
    }, 400);
  }
  var typingTimer;
  $('#username').on('keyup', function () {
    clearTimeout(typingTimer);
    var name = $(this).val().trim();
    var $status = $('#status');
    if (name.length < 3) { $status.slideUp(); return; }
    $status.removeClass('available taken').addClass('checking')
           .text('Checking...').slideDown();
    typingTimer = setTimeout(function () {
      checkUsername(name, function (data) {
        $status.removeClass('checking');
        if (data.available) {
          $status.removeClass('taken').addClass('available').text('"' + name + '" is available!');
        } else {
          $status.removeClass('available').addClass('taken').text('"' + name + '" is already taken.');
        }
        $status.hide().slideDown();
      });
    }, 300);
  });
});
</script>
</body>
</html>

<!DOCTYPE html>
<html lang="en">

<style>
  body { font-family: Arial; max-width: 500px; margin: 40px auto; padding: 0 20px; }
  h2 { color: #333; }
  #search { width: 100%; padding: 12px; box-sizing: border-box; font-size: 15px;
            border: 1px solid #ccc; border-radius: 4px; }
  #results { margin-top: 10px; }
  .product { display: none; padding: 12px; background: #f4f4f4; margin: 6px 0;
             border-radius: 4px; border-left: 4px solid #2196F3; }
  .no-results { padding: 12px; color: #c0392b; font-style: italic; }
</style>
</head>
<body>
  <h2>Product Search</h2>
  <input type="text" id="search" placeholder="Search for products...">
  <div id="results"></div>
<script>
$(document).ready(function () {
  var allProducts = [
    { name: 'iPhone 15 Pro', category: 'Mobiles' },
    { name: 'Samsung Galaxy S24', category: 'Mobiles' },
    { name: 'OnePlus 12', category: 'Mobiles' },
    { name: 'Dell XPS 15', category: 'Laptops' },
    { name: 'MacBook Pro M3', category: 'Laptops' },
    { name: 'HP Spectre x360', category: 'Laptops' },
    { name: 'Sony WH-1000XM5', category: 'Headphones' },
    { name: 'AirPods Pro', category: 'Headphones' },
    { name: 'iPad Air', category: 'Tablets' },
    { name: 'Samsung Tab S9', category: 'Tablets' }
  ];
  function fetchProducts(query, callback) {
    setTimeout(function () {
      callback(allProducts.filter(function (p) {
        return p.name.toLowerCase().indexOf(query.toLowerCase()) !== -1 ||
               p.category.toLowerCase().indexOf(query.toLowerCase()) !== -1;
      }));
    }, 150);
  }
  $('#search').on('keyup', function () {
    var query = $(this).val().trim();
    var $results = $('#results');
    $results.empty();
    if (query === '') return;
    fetchProducts(query, function (data) {
      if (data.length === 0) {
        $results.html('<div class="no-results">No results found for "' + query + '"</div>');
      } else {
        data.forEach(function (p) {
          var $item = $('<div class="product"></div>')
            .html('<strong>' + p.name + '</strong> <em>(' + p.category + ')</em>');
          $results.append($item);
          $item.fadeIn(300);
        });
      }
    });
  });
});
</script>
</body>
</html>

var allUsers = null;
var search   = document.getElementById('search');
var drop     = document.getElementById('dropdown');
var load     = document.getElementById('loading');
var timer;
search.addEventListener('input', function () {
  clearTimeout(timer);
  var q = search.value.trim().toLowerCase();
  if (!q) { drop.style.display='none'; drop.innerHTML=''; return; }
  timer = setTimeout(function () { fetchAndFilter(q); }, 250);
});
function fetchAndFilter(query) {
  if (allUsers) { renderResults(filterUsers(allUsers, query)); return; }
  load.style.display = 'block';
  drop.style.display = 'none';
  var xhr = new XMLHttpRequest();
  xhr.open('GET', 'https://jsonplaceholder.typicode.com/users', true);
  xhr.onload = function () {
    load.style.display = 'none';
    if (xhr.status === 200) {
      allUsers = JSON.parse(xhr.responseText);
      renderResults(filterUsers(allUsers, query));
    } else {
      drop.innerHTML = '<div class="err">API error ' + xhr.status + '. Try again.</div>';
      drop.style.display = 'block';
    }
  };
  xhr.onerror = function () {
    load.style.display = 'none';
    drop.innerHTML = '<div class="err">Network error. Make sure you are online.</div>';
    drop.style.display = 'block';
  };
  xhr.send();
}
function filterUsers(users, query) {
  return users.filter(function (u) {
    return u.name.toLowerCase().indexOf(query) !== -1 ||
           u.username.toLowerCase().indexOf(query) !== -1 ||
           u.email.toLowerCase().indexOf(query) !== -1;
  });
}
function renderResults(matches) {
  if (!matches.length) {
    drop.innerHTML = '<div class="item"><em>No users found</em></div>';
  } else {
    drop.innerHTML = matches.map(function (u) {
      return '<div class="item"><strong>' + u.name + '</strong>' +
             '<br><small>@' + u.username + ' &middot; ' + u.email +
             ' &middot; ' + u.address.city + '</small></div>';
    }).join('');
  }
  drop.style.display = 'block';
}

  <div class="input-row">
    <input type="text" id="newTask" placeholder="Add a new task...">

  </div>

  <div id="error"></div>
  <div id="taskList"></div>
<script>
var API = 'https://jsonplaceholder.typicode.com/todos';
var list   = document.getElementById('taskList');
var errDiv = document.getElementById('error');
function showSpinner(on){ document.getElementById('spinner').style.display = on ? 'block' : 'none'; }
function showErr(msg){ errDiv.innerHTML = msg ? '<div class="err">'+msg+'</div>' : ''; }
function loadTasks() {
  showSpinner(true); showErr('');
  fetch(API + '?_limit=5')
    .then(function(r){ if(!r.ok) throw new Error('HTTP '+r.status); return r.json(); })
    .then(function(data){ list.innerHTML=''; data.forEach(renderTask); })
    .catch(function(e){ showErr('Failed to load tasks: '+e.message+'. Make sure you are online.'); })
    .finally(function(){ showSpinner(false); });
}
function renderTask(task) {
  var div = document.createElement('div');
  div.className = 'task';
  div.id = 'task-' + task.id;
  div.innerHTML = '<span>' + task.title + '</span>' +
                  '<button class="del" onclick="deleteTask('+task.id+')">Delete</button>';
  list.appendChild(div);
}
function addTask() {
  var title = document.getElementById('newTask').value.trim();
  if (!title) return;
  showSpinner(true); showErr('');
  fetch(API, {
    method: 'POST',
    headers: {'Content-Type':'application/json'},
    body: JSON.stringify({title:title, completed:false, userId:1})
  })
  .then(function(r){ if(!r.ok) throw new Error('HTTP '+r.status); return r.json(); })
  .then(function(newTask){
    newTask.id = Date.now();
    renderTask(newTask);
    document.getElementById('newTask').value = '';
  })
  .catch(function(e){ showErr('Failed to add task: '+e.message); })
  .finally(function(){ showSpinner(false); });
}
function deleteTask(id) {
  showSpinner(true); showErr('');
  fetch(API + '/' + id, {method:'DELETE'})
    .then(function(r){ if(!r.ok && r.status !== 404) throw new Error('HTTP '+r.status); })
    .then(function(){
      var el = document.getElementById('task-'+id);
      if (el) el.remove();
    })
    .catch(function(e){ showErr('Failed to delete: '+e.message); })
    .finally(function(){ showSpinner(false); });
}
loadTasks();
</script>
</body>
</html>

function loadStates(){

function escapeHtml(s){
  return String(s).replace(/[&<>"']/g,function(c){
    return {'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c];
  });
}
function addCommentToDOM(name, body, prepend){
  var div = document.createElement('div');
  div.className = 'comment';
  div.innerHTML = '<div class="name">'+escapeHtml(name)+'</div><div>'+escapeHtml(body)+'</div>';
  var list = document.getElementById('commentList');
  if (prepend) list.insertBefore(div, list.firstChild);
  else         list.appendChild(div);
  requestAnimationFrame(function(){ requestAnimationFrame(function(){ div.classList.add('show'); }); });
}
function loadComments(){

      data.forEach(function(c){ addCommentToDOM(c.name, c.body, false); });

    document.getElementById('name').value='';
    document.getElementById('body').value='';
  })
  .catch(function(e){ err.textContent='Failed to post: '+e.message; });
}

$(function(){
  function loadNews(){
    $('#newsFeed').empty();
    $('#spinner').show();
    $('#refresh').prop('disabled',true).text('Loading...');
    $.ajax({
      url: 'https://jsonplaceholder.typicode.com/posts?_limit=6',
      method: 'GET',
      timeout: 8000,

        $('#spinner').hide();
        data.forEach(function(item, i){
          var $div = $('<div class="news-item"></div>')
            .append('<h3>'+item.title+'</h3>')
            .append('<p>'+item.body+'</p>');
          $('#newsFeed').append($div);
          $div.delay(i*200).slideDown(400);
        });
      },
      error: function(xhr, status){
        $('#spinner').hide();
        var reason = status==='timeout' ? 'Request timed out.' : 'HTTP '+xhr.status;
        $('#newsFeed').html('<div class="err-msg">Failed to load: '+reason+'. Check your internet connection.</div>');
      },
      complete: function(){
        $('#refresh').prop('disabled',false).html('&#x21bb; Refresh');
      }
    });
  }
  $('#refresh').click(function(){
    $('#newsFeed').children().slideUp(300, function(){ $(this).remove(); });
    setTimeout(loadNews, 350);
  });
  loadNews();
});

  function insertCartItem(p){
    cartCount++;
    var uid='cart-'+cartCount;
    var $item=$('<div class="cart-item" id="'+uid+'"></div>')
      .append('<span>'+p.name+' &mdash; $'+p.price.toFixed(2)+'</span>')
      .append($('<button class="remove">Remove</button>').click(function(){ removeFromCart(uid,p.price); }));
    $item.hide().appendTo('#cartList').fadeIn(300);
    $('#cartEmpty').hide();
    total+=p.price;
    $('#total').text('Total: $'+total.toFixed(2));
  }
  function removeFromCart(uid,price){
    $('#'+uid).fadeOut(300,function(){
      $(this).remove();
      total-=price;
      $('#total').text('Total: $'+total.toFixed(2));
      if ($('#cartList').children().length===0) $('#cartEmpty').show();
    });
  }
});
</script>

────────────────────────────────────────────────────────────────────────────────

PROBLEM:
  Implement an infinite scroll feature that:
  - Loads more content as the user scrolls down
  - Fetches initial posts from JSONPlaceholder
  - Loads more when the user reaches the bottom
  - Uses fadeIn() for smooth content loading
  - Shows a "Loading..." indicator during fetch

FILE: infinite_scroll.html

EXECUTION:
  Save the file and open in any browser. Requires internet connection.

──────
<!DOCTYPE html>
<html lang="en">
<head>
<meta charset="UTF-8"><title>Infinite Scroll</title>
<script src="https://code.jquery.com/jquery-3.6.0.min.js"></script>
<style>
  body{font-family:Arial;max-width:700px;margin:0 auto;padding:20px;background:#f5f5f5;}
  h2{color:#333;text-align:center;}
  .post{display:none;background:white;padding:15px;margin:10px 0;border-radius:6px;
        box-shadow:0 1px 3px rgba(0,0,0,.1);border-left:4px solid #2196F3;}
  .post h3{margin:0 0 8px;color:#1976D2;}
  .post p{margin:0;color:#555;line-height:1.5;}
  #loader{text-align:center;padding:20px;font-weight:bold;color:#2196F3;display:none;}
  #end{text-align:center;padding:20px;color:#777;font-style:italic;display:none;}
  .err-msg{background:#f8d7da;color:#721c24;padding:12px;border-radius:4px;text-align:center;}
</style>
</head>
<body>
  <h2>Infinite Scroll Demo</h2>
  <div id="posts"></div>
  <div id="loader">Loading more posts...</div>
  <div id="end">&#10003; You've seen all posts!</div>
<script>
$(function(){
  var page=1, perPage=5, loading=false, done=false, totalPosts=100;
  function loadPosts(){
    if (loading||done) return;
    loading=true; $('#loader').show();
    $.ajax({
      url:'https://jsonplaceholder.typicode.com/posts',
      data:{_page:page,_limit:perPage},
      timeout:8000,
      success:function(data){
        $('#loader').hide(); loading=false;
        if (!data||data.length===0){ done=true; $('#end').show(); return; }
        data.forEach(function(p){
          var $div=$('<div class="post"></div>')
            .append('<h3>'+p.title+'</h3>')
            .append('<p>'+p.body+'</p>');
          $('#posts').append($div);
          $div.fadeIn(400);
        });
        page++;
        if (page*perPage>totalPosts){ done=true; $('#end').show(); }
      },
      error:function(xhr,status){
        $('#loader').hide(); loading=false;
        var msg=status==='timeout'?'Request timed out':'HTTP error '+xhr.status;
        $('#posts').append('<div class="err-msg">Failed to load: '+msg+'. <a href="#" onclick="loadPosts();return false;">Retry</a></div>');
      }
    });
  }
  $(window).on('scroll',function(){
    if ($(window).scrollTop()+$(window).height()>=$(document).height()-150){
      loadPosts();
    }
  });
  loadPosts();
});
</script>
</body>
</html>

<script>
var poll = {
  id: 'poll-frontend-2024',
  question: "What's your favourite frontend technology?",
  options: [
    {id:'react',   text:'React',   votes:47},
    {id:'vue',     text:'Vue.js',  votes:32},
    {id:'angular', text:'Angular', votes:18},
    {id:'svelte',  text:'Svelte',  votes:24},
    {id:'jquery',  text:'jQuery',  votes:39}
  ]
};
$(function(){
  $('#question').text(poll.question);
  var voted = localStorage.getItem(poll.id);
  if (voted){
    showResults(true);
  } else {
    poll.options.forEach(function(o){
      $('<div class="option"></div>').text(o.text).data('id',o.id).appendTo('#optionsBox');
    });
  }
  $(document).on('click','.option',function(){
    var optId = $(this).data('id');
    submitVote(optId);
  });

function submitVote(optionId){
  $('.option').css({pointerEvents:'none',opacity:.6});
  $('#submitting').show();
  setTimeout(function(){
    poll.options.forEach(function(o){ if (o.id===optionId) o.votes++; });
    localStorage.setItem(poll.id, optionId);
    $('#submitting').hide();
    $('#optionsBox').slideUp(400,function(){
      $(this).empty();
      $('#voted').slideDown(300);
      showResults(false);
    });
  }, 600);
}
function showResults(immediate){
  var total = poll.options.reduce(function(s,o){ return s+o.votes; },0);
  var html  = '';
  poll.options.forEach(function(o){
    var pct = total===0 ? 0 : Math.round(o.votes/total*100);
    html += '<div class="bar-row">' +
            '<div class="bar-label"><span>'+o.text+'</span><span>'+o.votes+' votes ('+pct+'%)</span></div>' +
            '<div class="bar-wrap"><div class="fill" data-pct="'+pct+'"></div></div>' +
            '</div>';
  });
  $('#results').html(html);
  if (immediate){ $('#results').show(); $('#voted').show(); }
  else          { $('#results').slideDown(400); }
  $('#resetBtn').show();
  setTimeout(function(){
    $('.fill').each(function(){ $(this).css('width',$(this).data('pct')+'%'); });
  }, 150);
}
function resetVote(){
  localStorage.removeItem(poll.id);
  location.reload();
}
</script>
</body>
</html>
